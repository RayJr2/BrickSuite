cmake_minimum_required(VERSION 3.22)
project(GeogramRepairProof LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GEOGRAM_SOURCE_DIR "" CACHE PATH "Unmodified Geogram 1.10.0 release source")
if(NOT EXISTS "${GEOGRAM_SOURCE_DIR}/src/lib/geogram/mesh/mesh.h")
    message(FATAL_ERROR "Set GEOGRAM_SOURCE_DIR to the extracted official release")
endif()
foreach(feature GRAPHICS LEGACY_NUMERICS HLBFGS TETGEN TRIANGLE LUA FPG TBB EXPLORAGRAM)
    set(GEOGRAM_WITH_${feature} OFF CACHE BOOL "Disabled in isolated proof" FORCE)
endforeach()
set(GEOGRAM_LIB_ONLY ON CACHE BOOL "No bundled applications" FORCE)
add_subdirectory("${GEOGRAM_SOURCE_DIR}" geogram)
add_executable(geogram_repair repair.cpp)
target_link_libraries(geogram_repair PRIVATE geogram)
enable_testing()
add_test(NAME geogram_closed_tetrahedron COMMAND geogram_repair --self-test)
